cmake_minimum_required(VERSION 3.22.1)
project(HiFaceLinuxDemo VERSION 0.0.2 LANGUAGES CXX)
include(GNUInstallDirs)
set(HIFACE_DEFAULT_SDK "${CMAKE_CURRENT_SOURCE_DIR}/sdk")
if(NOT EXISTS "${HIFACE_DEFAULT_SDK}" AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/runtimes")
    string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" HIFACE_PROCESSOR)
    if(HIFACE_PROCESSOR MATCHES "^(aarch64|arm64)$")
        set(HIFACE_PROCESSOR arm64)
    elseif(HIFACE_PROCESSOR MATCHES "^(x86_64|amd64)$")
        set(HIFACE_PROCESSOR x86_64)
    else()
        message(FATAL_ERROR "Unsupported Linux processor: ${CMAKE_SYSTEM_PROCESSOR}")
    endif()
    set(HIFACE_DEFAULT_SDK "${CMAKE_CURRENT_SOURCE_DIR}/runtimes/${HIFACE_PROCESSOR}/sdk")
endif()
set(HIFACE_SDK_DIR "${HIFACE_DEFAULT_SDK}" CACHE PATH "Installed HiFace SDK directory")
if(NOT TARGET HiFace::HiFace)
    find_package(HiFace CONFIG REQUIRED HINTS "${HIFACE_SDK_DIR}")
endif()
find_package(Threads REQUIRED)
add_library(hiface_demo_common STATIC src/common/image_decode.cpp src/common/image_search.cpp src/common/gallery.cpp src/common/synthetic_data.cpp src/desktop/settings.cpp src/desktop/frame.cpp)
target_compile_features(hiface_demo_common PUBLIC cxx_std_17)
target_include_directories(hiface_demo_common PUBLIC src/common src/desktop PRIVATE vendor)
target_link_libraries(hiface_demo_common PUBLIC HiFace::HiFace)
add_executable(hiface_demo src/common/main.cpp)
target_link_libraries(hiface_demo PRIVATE hiface_demo_common)
target_link_options(hiface_demo PRIVATE "$<$<CONFIG:Release>:-s>")
set_target_properties(hiface_demo PROPERTIES INSTALL_RPATH "$ORIGIN/../sdk/lib")
install(TARGETS hiface_demo RUNTIME DESTINATION bin COMPONENT Demo)
option(HIFACE_BUILD_LINUX_DESKTOP "Build the X11/V4L2 desktop camera Demo" ON)
set(HIFACE_WORKFLOW_SOURCES src/desktop/workflow.cpp src/desktop/app.cpp src/desktop/camera.cpp)
if(HIFACE_BUILD_LINUX_DESKTOP)
    add_executable(hiface_desktop src/desktop/main.cpp src/desktop/window.cpp src/desktop/ui.cpp src/desktop/scene.cpp ${HIFACE_WORKFLOW_SOURCES})
    target_include_directories(hiface_desktop PRIVATE vendor)
    target_link_libraries(hiface_desktop PRIVATE hiface_demo_common Threads::Threads ${CMAKE_DL_LIBS})
    target_link_options(hiface_desktop PRIVATE "$<$<CONFIG:Release>:-s>")
    set_target_properties(hiface_desktop PROPERTIES INSTALL_RPATH "$ORIGIN/../sdk/lib")
    install(TARGETS hiface_desktop RUNTIME DESTINATION bin COMPONENT Demo)
endif()
install(FILES CMakeLists.txt README.md README.zh-CN.md DESTINATION . COMPONENT Demo)
install(DIRECTORY src/ DESTINATION src COMPONENT Demo)
install(DIRECTORY vendor/ DESTINATION vendor COMPONENT Demo)
install(PROGRAMS run-demo.sh self-test.sh runtime.sh DESTINATION . COMPONENT Demo)
set(HIFACE_DEMO_FIXTURE "${CMAKE_CURRENT_SOURCE_DIR}/assets/features_10000.json" CACHE FILEPATH "Synthetic Linux Demo features")
install(FILES "${HIFACE_DEMO_FIXTURE}" DESTINATION assets COMPONENT Demo)
